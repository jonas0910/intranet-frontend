// Dropdown con ngx-bootstrap
import {{ 'BsDropdownModule }' }} from 'ngx-bootstrap/dropdown';

// En el template:
<div class="dropdown" #dropdownExample="ngbDropdown">
  <button class="btn btn-primary dropdown-toggle" 
          type="button" 
          [ngbDropdownToggle]="dropdownRef">
    Dropdown
  </button>
  <div class="dropdown-menu" 
       #dropdownRef="ngbDropdownMenu">
    <button class="dropdown-item" type="button">Opción 1</button>
  </div>
</div>

// Modal con ngx-bootstrap  
import {{ 'BsModalService, BsModalRef }' }} from 'ngx-bootstrap/modal';

this.modalRef = this.modalService.show(MyModalComponent);

// Tooltip con ngx-bootstrap
import {{ 'TooltipModule }' }} from 'ngx-bootstrap/tooltip';

  Botón con tooltip
</button>
